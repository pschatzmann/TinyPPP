196 void process() {
197 if (!_started || _io == nullptr) return;
198 uint8_t buf[128];
199 // bound the work per call so that we do not block for too long
200 for (int loops = 0; loops < 32; loops++) {
201 int avail = _io->available();
202 if (avail <= 0) break;
203 size_t n = _io->readBytes(
204 buf, (size_t)avail < sizeof(buf) ? (size_t)avail : sizeof(buf));
205 if (n == 0) break;
206 for (size_t i = 0; i < n; i++) {
207 if (_dec.add(buf[i])) handleFrame(_dec.frame(), _dec.frameLen());
208 }
209 }
210 processTimers(millis());
211 }

437 void sendCP(uint16_t proto, uint8_t code, uint8_t id, const uint8_t *data,
438 size_t len) {
439 uint8_t pkt[4 + 128];
440 if (len > sizeof(pkt) - 4) len = sizeof(pkt) - 4;
441 pkt[0] = code;
442 pkt[1] = id;
443 put16(pkt + 2, (uint16_t)(len + 4));
444 if (len) memcpy(pkt + 4, data, len);
445 sendFrame(proto, pkt, len + 4);

450 void handleFrame(uint8_t *frame, size_t len) {
451 _stats.rxFrames++;
452 // address and control field may be compressed (ACFC)
453 if (len >= 2 && frame[0] == HDLC_ALL_STATIONS && frame[1] == HDLC_UI) {
454 frame += 2;
455 len -= 2;
456 }
457 if (len < 1) return;
458 // protocol field may be compressed (PFC): odd first byte
459 uint16_t proto;
460 if (frame[0] & 0x01) {
461 proto = frame[0];
462 frame += 1;
463 len -= 1;
464 } else {
465 if (len < 2) return;
466 proto = get16(frame);
467 frame += 2;
468 len -= 2;
469 }
470
471 switch (proto) {
472 case PPP_IP:
473 if (isUp()) {
474 _stats.rxIP++;
475 onIPv4Packet(frame, len);
476 } else {
477 _stats.rxDropped++;
478 }
479 return;
480 case PPP_LCP:
481 handleCP(_lcp, frame, len);
482 return;
483 case PPP_IPCP:
484 if (_lcp.state == CPState::Opened && _phase >= PPPPhase::Network) {
485 handleCP(_ipcp, frame, len);
486 } else {
487 _stats.rxDropped++;
488 }
489 return;
490 case PPP_PAP:
491 if (_lcp.state == CPState::Opened) handlePap(frame, len);
492 return;
493 default:
494 _stats.rxDropped++;
495 // unsupported protocol (CCP, IPV6CP, ...)
496 if (_lcp.state == CPState::Opened) sendProtocolReject(proto, frame, len);
497 return;
498 }
499 }

510 void handleCP(ControlProtocol &cp, const uint8_t *pkt, size_t len) {
511 if (len < 4) return;
512 uint8_t code = pkt[0];
513 uint8_t id = pkt[1];
514 size_t plen = get16(pkt + 2);
515 if (plen < 4 || plen > len) return; // malformed
516 const uint8_t *data = pkt + 4;
517 size_t dlen = plen - 4;
518 bool isLcp = cp.proto == PPP_LCP;
519
520 if (cp.state == CPState::Initial) {
521 // not started (e.g. IPCP before LCP is up or after stop())
522 return;
523 }
524
525 switch (code) {
526 case CONF_REQ:
527 receiveConfigRequest(cp, id, data, dlen);
528 break;
529 case CONF_ACK:
530 if (id != cp.id) return;
531 receiveConfigAck(cp);
532 break;
533 case CONF_NAK:
534 case CONF_REJ:
535 if (id != cp.id) return;
536 if (isLcp) {
537 lcpApplyNakRej(code, data, dlen);
538 } else {
539 ipcpApplyNakRej(code, data, dlen);
540 }
541 receiveConfigNak(cp);
542 break;
543 case TERM_REQ:
544 logf("%s terminate request", isLcp ? "LCP" : "IPCP");
545 sendCP(cp.proto, TERM_ACK, id, nullptr, 0);
546 if (isLcp) {
547 lcpDown();
548 // wait for the peer to come back
549 _lcp.state = CPState::Stopped;
550 if (!_passive) armTimer(_lcp, TINYPPP_RESTART_MS);
551 } else {
552 if (cp.state == CPState::Opened) ipcpDown();
553 cp.state = CPState::Stopped;
554 armTimer(cp, TINYPPP_RESTART_MS);
555 }
556 break;
557 case TERM_ACK:
558 if (cp.state == CPState::Opened) {
559 if (isLcp) {
560 lcpDown();
561 lcpOpen();
562 } else {
563 ipcpDown();
564 ipcpOpen();
565 }
566 } else if (cp.state == CPState::Closing) {
567 cp.state = CPState::Stopped;
568 if (isLcp && !_passive) armTimer(cp, TINYPPP_RESTART_MS);
569 }
570 break;
571 case CODE_REJ:
572 logf("code reject");
573 break;
574 case PROT_REJ:
575 if (isLcp && dlen >= 2) {
576 uint16_t p = get16(data);
577 logf("peer rejected protocol 0x%04x", p);
578 if (p == PPP_IPCP) {
579 ipcpDown();
580 _ipcp.state = CPState::Initial;
581 _ipcp.timer = 0;
582 }
583 }
584 break;
585 case ECHO_REQ:
586 if (isLcp && cp.state == CPState::Opened) {
587 uint8_t reply[64];
588 size_t n = dlen < sizeof(reply) ? dlen : sizeof(reply);
589 memcpy(reply, data, n);
590 if (n >= 4) memcpy(reply, &_magic, 4);
591 sendCP(PPP_LCP, ECHO_REP, id, reply, n);
592 }
593 break;
594 case ECHO_REP:
595 _echoPending = 0;
596 break;
597 case DISCARD_REQ:
598 break;
599 default: {
600 // unknown code: Code-Reject
601 sendCP(cp.proto, CODE_REJ, ++cp.id, pkt, plen);
602 break;
603 }
604 }
605 }

609 void receiveConfigRequest(ControlProtocol &cp, uint8_t id,
610 const uint8_t *data, size_t len) {
611 uint8_t reply[128];
612 size_t replyLen = 0;
613 uint8_t code = cp.proto == PPP_LCP
614 ? lcpCheckRequest(data, len, reply, replyLen)
615 : ipcpCheckRequest(data, len, reply, replyLen);
616 if (code == 0) return; // malformed
617 bool ack = code == CONF_ACK;
618
619 switch (cp.state) {
620 case CPState::Opened:
621 // renegotiation
622 layerDown(cp);
623 sendConfigRequest(cp);
624 cp.state = ack ? CPState::AckSent : CPState::ReqSent;
625 break;
626 case CPState::Stopped:
627 case CPState::Closing:
628 sendConfigRequest(cp);
629 cp.state = ack ? CPState::AckSent : CPState::ReqSent;
630 break;
631 case CPState::ReqSent:
632 case CPState::AckSent:
633 cp.state = ack ? CPState::AckSent : CPState::ReqSent;
634 break;
635 case CPState::AckRcvd:
636 if (ack) {
637 sendCP(cp.proto, code, id, reply, replyLen);
638 cp.state = CPState::Opened;
639 layerUp(cp);
640 return;
641 }
642 break;
643 default:
644 break;
645 }
646 sendCP(cp.proto, code, id, reply, replyLen);
647 }

649 void receiveConfigAck(ControlProtocol &cp) {
650 cp.retries = 0;
651 switch (cp.state) {
652 case CPState::ReqSent:
653 cp.state = CPState::AckRcvd;
654 armTimer(cp);
655 break;
656 case CPState::AckSent:
657 cp.state = CPState::Opened;
658 cp.timer = 0;
659 layerUp(cp);
660 break;
661 case CPState::AckRcvd:
662 // crossed connection
663 sendConfigRequest(cp);
664 cp.state = CPState::ReqSent;
665 break;
666 case CPState::Opened:
667 layerDown(cp);
668 sendConfigRequest(cp);
669 cp.state = CPState::ReqSent;
670 break;
671 default:
672 break;
673 }
674 }

729 void lcpTimeout() {
730 switch (_lcp.state) {
731 case CPState::ReqSent:
732 case CPState::AckRcvd:
733 case CPState::AckSent:
734 // keep on trying forever: the host might start pppd later
735 if (++_lcp.retries > 10) _lcp.retries = 10;
736 sendConfigRequest(_lcp);
737 if (_lcp.state == CPState::AckRcvd) _lcp.state = CPState::ReqSent;
738 // back off to reduce noise on the line while nobody is listening
739 if (_lcp.retries >= 10) armTimer(_lcp, 5 * TINYPPP_RESTART_MS);
740 break;
741 case CPState::Stopped:
742 case CPState::Closing:
743 if (!_passive) lcpOpen();
744 break;
745 default:
746 break;
747 }

778 size_t n = 0;
779 if (_reqAccm) {
780 p[n++] = LCP_ACCM;
781 p[n++] = 6;
782 memset(p + n, 0, 4); // nothing needs to be escaped

794 void lcpApplyNakRej(uint8_t code, const uint8_t *p, size_t len) {
795 size_t pos = 0;
796 while (pos + 2 <= len) {
797 uint8_t type = p[pos];
798 uint8_t olen = p[pos + 1];
799 if (olen < 2 || pos + olen > len) return;
800 if (type == LCP_ACCM) {
801 _reqAccm = false; // keep the default map
802 } else if (type == LCP_MAGIC) {
803 if (code == CONF_REJ) {
804 _reqMagic = false;
805 } else {
806 _magic = newMagic();
807 }
808 }
809 pos += olen;
810 }
811 }

813 uint8_t lcpCheckRequest(const uint8_t *p, size_t len, uint8_t *reply,
814 size_t &replyLen) {
815 uint8_t naks[32];
816 size_t nakLen = 0;
817 size_t rejLen = 0;
818 uint16_t mru = 1500;
819 uint32_t accm = 0xFFFFFFFF;
820 bool needPap = false;
821
822 size_t pos = 0;
823 while (pos < len) {
824 if (pos + 2 > len) return 0;
825 uint8_t type = p[pos];
826 uint8_t olen = p[pos + 1];
827 if (olen < 2 || pos + olen > len) return 0;
828 const uint8_t *opt = p + pos;
829 bool reject = false;
830 switch (type) {
831 case LCP_MRU:
832 if (olen != 4) {
833 reject = true;
834 } else {
835 mru = get16(opt + 2);
836 if (mru > TINYPPP_MRU) mru = TINYPPP_MRU;
837 }
838 break;
839 case LCP_ACCM:
840 if (olen != 6) {
841 reject = true;
842 } else {
843 accm = (uint32_t)opt[2] << 24 | (uint32_t)opt[3] << 16 |
844 (uint32_t)opt[4] << 8 | opt[5];
845 }
846 break;
847 case LCP_AUTH:
848 if (olen >= 4 && get16(opt + 2) == PPP_PAP) {
849 needPap = true;
850 } else if (nakLen + 4 <= sizeof(naks)) {
851 // we only support PAP
852 naks[nakLen++] = LCP_AUTH;
853 naks[nakLen++] = 4;
854 put16(naks + nakLen, PPP_PAP);
855 nakLen += 2;
856 }
857 break;
858 case LCP_MAGIC:
859 if (olen != 6) {
860 reject = true;
861 } else if (_reqMagic && memcmp(opt + 2, &_magic, 4) == 0 &&
862 nakLen + 6 <= sizeof(naks)) {
863 // looped back line or same magic: suggest another value
864 uint32_t m = newMagic();
865 naks[nakLen++] = LCP_MAGIC;
866 naks[nakLen++] = 6;
867 memcpy(naks + nakLen, &m, 4);
868 nakLen += 4;
869 }
870 break;
871 case LCP_PFC:
872 case LCP_ACFC:
873 // we accept compressed frames but never send them
874 if (olen != 2) reject = true;
875 break;
876 default:
877 reject = true;
878 break;
879 }
880 if (reject && rejLen + olen <= 128) {
881 memcpy(reply + rejLen, opt, olen);
882 rejLen += olen;
883 }
884 pos += olen;
885 }
886
887 if (rejLen > 0) {
888 replyLen = rejLen;
889 return CONF_REJ;
890 }
891 if (nakLen > 0) {
892 memcpy(reply, naks, nakLen);
893 replyLen = nakLen;
894 return CONF_NAK;
895 }
896 // Ack: commit the peer's options
897 _peerMRU = mru;
898 _peerAccm = accm;
899 _needPap = needPap;
900 size_t n = len < 128 ? len : 128;
901 memcpy(reply, p, n);
902 replyLen = n;
903 return CONF_ACK;
904 }

908 void papSend() {
909 uint8_t data[2 + sizeof(_user) + sizeof(_password)];
910 size_t ul = strlen(_user);
911 size_t pl = strlen(_password);
912 size_t n = 0;
913 data[n++] = (uint8_t)ul;
914 memcpy(data + n, _user, ul);
915 n += ul;
916 data[n++] = (uint8_t)pl;
917 memcpy(data + n, _password, pl);
918 n += pl;
919 uint8_t pkt[4 + sizeof(data)];
920 pkt[0] = 1; // Authenticate-Request
921 pkt[1] = ++_papId;
922 put16(pkt + 2, (uint16_t)(n + 4));
923 memcpy(pkt + 4, data, n);
924 sendFrame(PPP_PAP, pkt, n + 4);
925 _papTimer = millis() + TINYPPP_RESTART_MS;
926 if (_papTimer == 0) _papTimer = 1;
927 }

930 if (_phase != PPPPhase::Authenticate) return;
931 if (++_papRetries > 10) {
932 logf("PAP: no response");
933 lcpClose();
934 return;
935 }
936 papSend();

939 void handlePap(const uint8_t *pkt, size_t len) {
940 if (len < 4 || _phase != PPPPhase::Authenticate) return;
941 uint8_t code = pkt[0];
942 if (pkt[1] != _papId) return;
943 if (code == 2) { // Authenticate-Ack
944 logf("PAP: authenticated");
945 _papTimer = 0;
946 setPhase(PPPPhase::Network);
947 ipcpOpen();
948 } else if (code == 3) { // Authenticate-Nak
949 logf("PAP: authentication failed");
950 _papTimer = 0;
951 lcpClose();
952 }

1067 uint8_t ipcpCheckRequest(const uint8_t *p, size_t len, uint8_t *reply,
1068 size_t &replyLen) {
1069 uint8_t naks[32];
1070 size_t nakLen = 0;
1071 size_t rejLen = 0;
1072 ip4_t peer = 0;
1073
1074 size_t pos = 0;
1075 while (pos < len) {
1076 if (pos + 2 > len) return 0;
1077 uint8_t type = p[pos];
1078 uint8_t olen = p[pos + 1];
1079 if (olen < 2 || pos + olen > len) return 0;
1080 const uint8_t *opt = p + pos;
1081 ip4_t v = 0;
1082 if (olen == 6) memcpy(&v, opt + 2, 4);
1083 bool reject = false;
1084 switch (type) {
1085 case IPCP_ADDRESS:
1086 if (olen != 6) {
1087 reject = true;
1088 } else if (v == 0 || (v == _ourIP && _ourIP != 0)) {
1089 // the peer wants an address from us
1090 if (_cfgPeer != 0 && _cfgPeer != _ourIP) {
1091 putAddr(naks, nakLen, IPCP_ADDRESS, _cfgPeer);
1092 } else {
1093 reject = true;
1094 }
1095 } else {
1096 peer = v;
1097 }
1098 break;
1099 case IPCP_DNS1:
1100 case IPCP_DNS2: {
1101 ip4_t offer = type == IPCP_DNS1 ? _cfgOfferDns1 : _cfgOfferDns2;
1102 if (olen != 6 || offer == 0) {
1103 reject = true;
1104 } else if (v != offer) {
1105 putAddr(naks, nakLen, type, offer);
1106 }
1107 break;
1108 }
1109 default:
1110 // IP compression (VJ) etc. are not supported
1111 reject = true;
1112 break;
1113 }
1114 if (reject && rejLen + olen <= 128) {
1115 memcpy(reply + rejLen, opt, olen);
1116 rejLen += olen;
1117 }
1118 pos += olen;
1119 }
1120
1121 if (rejLen > 0) {
1122 replyLen = rejLen;
1123 return CONF_REJ;
1124 }
1125 if (nakLen > 0) {
1126 memcpy(reply, naks, nakLen);
1127 replyLen = nakLen;
1128 return CONF_NAK;
1129 }
1130 _peerIP = peer != 0 ? peer : _cfgPeer;
1131 size_t n = len < 128 ? len : 128;
1132 memcpy(reply, p, n);
1133 replyLen = n;
1134 return CONF_ACK;
1135 }
