52 bool begin(Stream &io) {
53 if (_netifAdded) end();
54 _io = &io;
55
56 ip4_addr_t any;
57 ip4_addr_set_zero(&any);
58 memset(&_netif, 0, sizeof(_netif));
59 if (!netif_add(&_netif, &any, &any, &any, this, netifInit, ip4_input)) {
60 return false;
61 }
62 _netifAdded = true;
63 if (_addNetifCB) _addNetifCB(&_netif);
64
65 // the engine is started from the lwIP context in the packet handler
66 _cmd = Cmd::Start;
67 _phID = __addEthernetPacketHandler([this] { this->poll(); });
68 return true;
69 }
70
72 void end() {
73 if (!_netifAdded) return;
74 if (_phID >= 0) {
75 _cmd = Cmd::Stop;
76 // wait for the handler to send the LCP Terminate-Request
77 uint32_t start = millis();
78 while (_cmd != Cmd::None && millis() - start < 1000) delay(1);
79 __removeEthernetPacketHandler(_phID);
80 _phID = -1;
81 }
82 if (_removeNetifCB) _removeNetifCB(&_netif);
83 netif_remove(&_netif);
84 _netifAdded = false;
85 }

189 static err_t netifInit(struct netif *netif) {
190 netif->name[0] = 'p';
191 netif->name[1] = 'p';
192 netif->mtu = 1500;
193 netif->flags = 0; // point-to-point: no ARP, no broadcast
194 netif->output = output4;
195#if LWIP_IPV6
196 netif->output_ip6 = output6;
197#endif
198 return ERR_OK;
199 }

201 static err_t output4(struct netif *netif, struct pbuf *p,
202 const ip4_addr_t *addr) {
203 (void)addr;
204 TinyPPP_RP2040 *self = (TinyPPP_RP2040 *)netif->state;
205 if (!self->beginIPv4()) return ERR_CONN;
206 for (struct pbuf *q = p; q != nullptr; q = q->next) {
207 self->writeIPv4((const uint8_t *)q->payload, q->len);
208 if (q->len == q->tot_len) break;
209 }
210 return self->endIPv4() ? ERR_OK : ERR_IF;
211 }

226 void onLinkUp() override {
227 ip4_addr_t ip, mask, gw;
228 ip4_addr_set_u32(&ip, PPPEngine::localIP());
229 ip4_addr_set_u32(&gw, peerIP());
230 ip4_addr_set_u32(&mask, 0xFFFFFFFF);
231 netif_set_addr(&_netif, &ip, &mask, &gw);
232 _netif.mtu = mtu();
233 netif_set_link_up(&_netif);
234 netif_set_up(&_netif);
235 if (dns1() != 0) {
236 ip_addr_t dns;
237 ip_addr_set_ip4_u32(&dns, dns1());
238 dns_setserver(0, &dns);
239 }
240 if (dns2() != 0) {
241 ip_addr_t dns;
242 ip_addr_set_ip4_u32(&dns, dns2());
243 dns_setserver(1, &dns);
244 }
245 if (_default) netif_set_default(&_netif);
246 _up = true;
247 if (_stateCallback) _stateCallback(true, _stateRef);
248 }
249
250 void onLinkDown() override {
251 _up = false;
252 netif_set_down(&_netif);
253 netif_set_link_down(&_netif);
254 ip4_addr_t any;
255 ip4_addr_set_zero(&any);
256 netif_set_addr(&_netif, &any, &any, &any);
257 if (_stateCallback) _stateCallback(false, _stateRef);
258 }
