50 bool begin(Stream &io) {
51 if (_netifAdded) end();
52 _io = &io;
53
54 ip4_addr_t any;
55 ip4_addr_set_zero(&any);
56 memset(&_netif, 0, sizeof(_netif));
57 if (!netif_add(&_netif, &any, &any, &any, this, netifInit, ip4_input)) {
58 return false;
59 }
60 _netifAdded = true;
61 if (_addNetifCB) _addNetifCB(&_netif);
62
63 // the engine is started from the lwIP context in the packet handler
64 _cmd = Cmd::Start;
65 _phID = __addEthernetPacketHandler([this] { this->poll(); });
66 return true;
67 }
68
70 void end() {
71 if (!_netifAdded) return;
72 if (_phID >= 0) {
73 _cmd = Cmd::Stop;
74 // wait for the handler to send the LCP Terminate-Request
75 uint32_t start = millis();
76 while (_cmd != Cmd::None && millis() - start < 1000) delay(1);
77 __removeEthernetPacketHandler(_phID);
78 _phID = -1;
79 }
80 if (_removeNetifCB) _removeNetifCB(&_netif);
81 netif_remove(&_netif);
82 _netifAdded = false;
83 }

188 netif->name[0] = 'p';

199 static err_t output4(struct netif *netif, struct pbuf *p,
200 const ip4_addr_t *addr) {
201 (void)addr;
202 TinyPPP_RP2040 *self = (TinyPPP_RP2040 *)netif->state;
203 if (!self->beginIPv4()) return ERR_CONN;
204 for (struct pbuf *q = p; q != nullptr; q = q->next) {
205 self->writeIPv4((const uint8_t *)q->payload, q->len);
206 if (q->len == q->tot_len) break;
207 }
208 return self->endIPv4() ? ERR_OK : ERR_IF;
209 }

218 struct pbuf *p = pbuf_alloc(PBUF_RAW, (u16_t)len, PBUF_POOL);
219 if (p == nullptr) return;

224 void onLinkUp() override {
225 ip4_addr_t ip, mask, gw;
226 ip4_addr_set_u32(&ip, PPPEngine::localIP());
227 ip4_addr_set_u32(&gw, peerIP());
228 ip4_addr_set_u32(&mask, 0xFFFFFFFF);
229 netif_set_addr(&_netif, &ip, &mask, &gw);
230 _netif.mtu = mtu();
231 netif_set_link_up(&_netif);
232 netif_set_up(&_netif);
233 if (dns1() != 0) {
234 ip_addr_t dns;
235 ip_addr_set_ip4_u32(&dns, dns1());
236 dns_setserver(0, &dns);
237 }
238 if (dns2() != 0) {
239 ip_addr_t dns;
240 ip_addr_set_ip4_u32(&dns, dns2());
241 dns_setserver(1, &dns);
242 }
243 if (_default) netif_set_default(&_netif);
244 _up = true;
245 if (_stateCallback) _stateCallback(true, _stateRef);
246 }
247
248 void onLinkDown() override {
249 _up = false;
250 netif_set_down(&_netif);
251 netif_set_link_down(&_netif);
252 ip4_addr_t any;
253 ip4_addr_set_zero(&any);
254 netif_set_addr(&_netif, &any, &any, &any);
255 if (_stateCallback) _stateCallback(false, _stateRef);
256 }
