22inline uint16_t fcs16Update(uint16_t fcs, uint8_t b) {
23 static const uint16_t table[16] = {
24 0x0000, 0x1081, 0x2102, 0x3183, 0x4204, 0x5285, 0x6306, 0x7387,
25 0x8408, 0x9489, 0xA50A, 0xB58B, 0xC60C, 0xD68D, 0xE70E, 0xF78F};
26 fcs = (fcs >> 4) ^ table[(fcs ^ b) & 0x0F];
27 fcs = (fcs >> 4) ^ table[(fcs ^ (b >> 4)) & 0x0F];
28 return fcs;
29}
