203 // esp_netif driver: output of the PPP stack (called from the lwIP task)
204 static esp_err_t transmit(void *handle, void *buffer, size_t len) {
205 TinyPPP_ESP32 *self = ((Driver *)handle)->self;
206 if (self->_io == nullptr) return ESP_FAIL;
207 const uint8_t *data = (const uint8_t *)buffer;
208 size_t done = 0;
209 for (int retry = 0; retry < 100 && done < len; retry++) {
210 size_t n = self->_io->write(data + done, len - done);
211 done += n;
212 if (n == 0) vTaskDelay(1);
213 }
214 return done == len ? ESP_OK : ESP_FAIL;
215 }

217 // reads the stream and feeds the PPP stack
218 static void rxTask(void *arg) {
219 TinyPPP_ESP32 *self = (TinyPPP_ESP32 *)arg;
220 uint8_t buffer[TINYPPP_RX_BUFFER];
221 while (self->_running) {
222 int avail = self->_io->available();
223 if (avail > 0) {
224 size_t n = self->_io->readBytes(
225 buffer, (size_t)avail < sizeof(buffer) ? (size_t)avail : sizeof(buffer));
226 if (n > 0) esp_netif_receive(self->_esp_netif, buffer, n, nullptr);
227 } else {
228 vTaskDelay(1);
229 }
230 // restart the negotiation after the link was lost
231 uint32_t restart = self->_restartAt;
232 if (restart != 0 && !self->_stopping &&
233 (int32_t)(millis() - restart) >= 0) {
234 self->_restartAt = 0;
235 self->log("restarting");
236 esp_netif_action_start(self->_esp_netif, nullptr, 0, nullptr);
237 }
238 }
239 self->_task = nullptr;
240 vTaskDelete(nullptr);
241 }

256 static void pppEventCb(void *arg, esp_event_base_t base, int32_t id,
257 void *data) {
258 TinyPPP_ESP32 *self = (TinyPPP_ESP32 *)arg;
259 // event data is a esp_netif_t** (except for NETIF_PPP_CONNECT_FAILED where
260 // ESP-IDF posts the netif struct itself: there is only one PPP slot anyway)
261 if (id != NETIF_PPP_CONNECT_FAILED) {
262 esp_netif_t *netif = data ? *(esp_netif_t **)data : nullptr;
263 if (netif != self->_esp_netif) return;
264 }
265 self->onPppEvent(id);

268 void onPppEvent(int32_t id) {
269 if (id == NETIF_PPP_ERRORNONE) {
270 log("connected");
271 if (!(getStatusBits() & ESP_NETIF_CONNECTED_BIT)) {
272 setStatusBits(ESP_NETIF_CONNECTED_BIT);
273 postEvent(ARDUINO_EVENT_PPP_CONNECTED);
274 }
275 } else if (id < NETIF_PP_PHASE_OFFSET) {
276 // error: the link is down
277 log("link down, error", id);
278 if (getStatusBits() & ESP_NETIF_CONNECTED_BIT) {
279 clearStatusBits(ESP_NETIF_CONNECTED_BIT);
280 postEvent(ARDUINO_EVENT_PPP_DISCONNECTED);
281 }
282 if (_stopping) {
283 xEventGroupSetBits(_events(), DEAD_BIT);
284 } else {
285 _restartAt = millis() + 1000;
286 }
287 } else if (id == NETIF_PPP_PHASE_DEAD) {
288 log("phase dead");
289 if (_stopping) xEventGroupSetBits(_events(), DEAD_BIT);
290 } else if (id == NETIF_PPP_PHASE_RUNNING) {
291 log("phase running");
292 } else if (id == NETIF_PPP_PHASE_ESTABLISH) {
293 log("phase establish");
294 } else if (id == NETIF_PPP_PHASE_NETWORK) {
295 log("phase network");
296 } else if (id == NETIF_PPP_CONNECT_FAILED) {
297 log("connect failed");
298 if (!_stopping) _restartAt = millis() + 2000;
299 }
300 }
