201 // esp_netif driver: output of the PPP stack (called from the lwIP task)
202 static esp_err_t transmit(void *handle, void *buffer, size_t len) {
203 TinyPPP_ESP32 *self = ((Driver *)handle)->self;
204 if (self->_io == nullptr) return ESP_FAIL;
205 const uint8_t *data = (const uint8_t *)buffer;
206 size_t done = 0;
207 for (int retry = 0; retry < 100 && done < len; retry++) {
208 size_t n = self->_io->write(data + done, len - done);
209 done += n;
210 if (n == 0) vTaskDelay(1);
211 }
212 return done == len ? ESP_OK : ESP_FAIL;
213 }

215 // reads the stream and feeds the PPP stack
216 static void rxTask(void *arg) {
217 TinyPPP_ESP32 *self = (TinyPPP_ESP32 *)arg;
218 uint8_t buffer[TINYPPP_RX_BUFFER];
219 while (self->_running) {
220 int avail = self->_io->available();
221 if (avail > 0) {
222 size_t n = self->_io->readBytes(
223 buffer, (size_t)avail < sizeof(buffer) ? (size_t)avail : sizeof(buffer));
224 if (n > 0) esp_netif_receive(self->_esp_netif, buffer, n, nullptr);
225 } else {
226 vTaskDelay(1);
227 }
228 // restart the negotiation after the link was lost
229 uint32_t restart = self->_restartAt;
230 if (restart != 0 && !self->_stopping &&
231 (int32_t)(millis() - restart) >= 0) {
232 self->_restartAt = 0;
233 self->log("restarting");
234 esp_netif_action_start(self->_esp_netif, nullptr, 0, nullptr);
235 }
236 }
237 self->_task = nullptr;
238 vTaskDelete(nullptr);
239 }

254 static void pppEventCb(void *arg, esp_event_base_t base, int32_t id,
255 void *data) {
256 TinyPPP_ESP32 *self = (TinyPPP_ESP32 *)arg;
257 // event data is a esp_netif_t** (except for NETIF_PPP_CONNECT_FAILED where
258 // ESP-IDF posts the netif struct itself: there is only one PPP slot anyway)
259 if (id != NETIF_PPP_CONNECT_FAILED) {

261 if (netif != self->_esp_netif) return;

266 void onPppEvent(int32_t id) {
267 if (id == NETIF_PPP_ERRORNONE) {
268 log("connected");
269 if (!(getStatusBits() & ESP_NETIF_CONNECTED_BIT)) {
270 setStatusBits(ESP_NETIF_CONNECTED_BIT);
271 postEvent(ARDUINO_EVENT_PPP_CONNECTED);
272 }
273 } else if (id < NETIF_PP_PHASE_OFFSET) {
274 // error: the link is down
275 log("link down, error", id);
276 if (getStatusBits() & ESP_NETIF_CONNECTED_BIT) {
277 clearStatusBits(ESP_NETIF_CONNECTED_BIT);
278 postEvent(ARDUINO_EVENT_PPP_DISCONNECTED);
279 }
280 if (_stopping) {
281 xEventGroupSetBits(_events(), DEAD_BIT);
282 } else {
283 _restartAt = millis() + 1000;
284 }
285 } else if (id == NETIF_PPP_PHASE_DEAD) {
286 log("phase dead");
287 if (_stopping) xEventGroupSetBits(_events(), DEAD_BIT);
288 } else if (id == NETIF_PPP_PHASE_RUNNING) {
289 log("phase running");
290 } else if (id == NETIF_PPP_PHASE_ESTABLISH) {
291 log("phase establish");
292 } else if (id == NETIF_PPP_PHASE_NETWORK) {
293 log("phase network");
294 } else if (id == NETIF_PPP_CONNECT_FAILED) {
295 log("connect failed");
296 if (!_stopping) _restartAt = millis() + 2000;
297 }
298 }
